# 添加处理二次反射的深度计算方法
def calculate_adjusted_depth(self, channel, depth, rail_type):
    """
    计算调整后的深度，处理二次反射
    
    Args:
        channel: 通道号
        depth: 原始计算深度
        rail_type: 轨型
        
    Returns:
        调整后的深度
    """
    # 只有斜70°通道（0-3）需要处理二次反射
    if 0 <= channel <= 3:
        if self.depth_display_mode == "V型":
            # 模式1：根据轨型确定最大深度，处理二次反射
            max_depth = self.max_depth_by_rail_type.get(rail_type, 39)  # 默认60轨
            if depth > max_depth:
                adjusted_depth = 2 * max_depth - depth
            else:
                adjusted_depth = depth
        else:  # "压缩"模式
            # 模式2：直接将深度*0.5
            adjusted_depth = depth * 0.5
        return adjusted_depth
    else:
        # 其他通道不处理
        return depth

### 3. 修改MainWindow类

**位置**：MainWindow类的\_\_init\_\_方法

**修改内容**：

```python
# 添加显示模式切换按钮
self.create_display_mode_buttons()

# 添加创建显示模式按钮的方法
def create_display_mode_buttons(self):
    """
    创建显示模式切换按钮
    """
    # 创建按钮组
    self.display_mode_group = QButtonGroup()
    
    # 创建按钮布局
    mode_layout = QHBoxLayout()
    
    # 创建V型模式按钮
    self.v_mode_btn = QRadioButton("V型")
    self.v_mode_btn.setChecked(True)  # 默认选中V型模式
    self.v_mode_btn.toggled.connect(self.on_display_mode_changed)
    self.display_mode_group.addButton(self.v_mode_btn)
    mode_layout.addWidget(self.v_mode_btn)
    
    # 创建压缩模式按钮
    self.compress_mode_btn = QRadioButton("压缩")
    self.compress_mode_btn.toggled.connect(self.on_display_mode_changed)
    self.display_mode_group.addButton(self.compress_mode_btn)
    mode_layout.addWidget(self.compress_mode_btn)
    
    # 添加到主布局
    self.main_layout.addLayout(mode_layout)

# 添加显示模式切换事件处理
def on_display_mode_changed(self):
    """
    显示模式切换事件处理
    """
    if self.v_mode_btn.isChecked():
        self.bscan_canvas.depth_display_mode = "V型"
    else:
        self.bscan_canvas.depth_display_mode = "压缩"
    
    # 更新B显图像
    if self.parser is not None:
        annotations = self.annotation_manager.get_annotations()
        self.bscan_canvas.update_bscan(self.parser, self.display_range, self.current_e, annotations)
```
